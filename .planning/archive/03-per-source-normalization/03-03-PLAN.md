```sas
/* Program: 03_prep_mdN.sas | Phase 3 | Requirements: PREP-01,PREP-02,PREP-05,PREP-06 */
options mprint nofmterr;
%let expected_nobs = <FROZEN COUNT FOR N>;

/* SECTION 0 paths (as above) */

/* SECTION 1 preconditions (%abort inside %macro only) */
%macro check_libname(lib=);
  %if %sysfunc(libref(&lib)) ne 0 %then %do; %put ERROR: LIBNAME &lib not assigned.; %abort cancel; %end;
%mend check_libname;
%macro check_dir(path=, label=);
  %if %sysfunc(fileexist(&path)) = 0 %then %do; %put ERROR: &label dir missing: &path; %abort cancel; %end;
%mend check_dir;
%check_libname(lib=src); %check_libname(lib=g);
%check_dir(path=&qc_path, label=qc); %check_dir(path=&logs_path, label=logs);

/* SECTION 2 exception report.
   Two counts, and BOTH are measured -- never write a hardcoded 0 (RESEARCH Pitfall 10).
     n_sent : literal 'NULL' sentinel strings in this source's character variables.
              Expected 0 for md1-md7 (only md8 has them). A nonzero result means this
              source was re-exported from Excel like md8 was -> ABORT.
     n_enc  : encoding-damaged Base_Procedure_1 rows -> FLAG ONLY (PCM-C-01).            */
proc sql noprint;
  select count(*) into :n_sent trimmed from src.master_data_N
  where strip(upcase(<CHARVAR1>)) = 'NULL'
     or strip(upcase(<CHARVAR2>)) = 'NULL'
     /* ... every character variable of mdN, from qc/03_charvars_all.txt ... */ ;

  select count(*) into :n_enc trimmed from src.master_data_N
  where not missing(Base_Procedure_1)
    and verify(Base_Procedure_1, ' !"#$%&' || "'" || '()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`abcdefghijklmnopqrstuvwxyz{|}~') > 0;
quit;
filename excf "&qc_path.\03_exceptions_mdN.txt";
data _null_;
  file excf;
  put "mdN Exception Report -- Run: %sysfunc(datetime(), datetime20.)";
  put "NULL sentinel strings in character variables (abort if nonzero): &n_sent";
  put "Encoding-damaged Base_Procedure_1 rows (flag only, PCM-C-01): &n_enc";
run;
filename excf clear;
%macro assert_zero(n=, msg=);
  %if &n > 0 %then %do; %put ERROR: PREP-02 VIOLATION -- &n &msg; %abort cancel; %end;
  %else %put NOTE: PREP-02 OK -- 0 &msg;
%mend assert_zero;
%assert_zero(n=&n_sent, msg=NULL sentinel strings in mdN (source may have been re-exported));
%put NOTE: PREP-02 mdN -- &n_enc encoding-damaged rows flagged (no abort, PCM-C-01).;

/* SECTION 3 LENGTH-before-SET copy to g.prep_mdN */
data g.prep_mdN;
  length
    PRECEDE_STUDY_ID $12
    /* EVERY character variable of mdN at its width from qc/03_charvars_all.txt */
    ;
  set src.master_data_N;
run;

/* SECTION 4 conversion log (zero conversions expected) */
proc sql noprint; select count(*) into :n_total trimmed from g.prep_mdN; quit;
filename cl "&logs_path.\03_conversions_mdN.txt";
data _null_;
  file cl;
  put "mdN Conversion Count Log -- Run: %sysfunc(datetime(), datetime20.)";
  put "Total rows: &n_total";
  put "Type conversions performed: 0 (mdN has no forced-char numerics)";
  put "NULL sentinels found (asserted zero pre-copy): &n_sent";
  put "Encoding-damaged rows flagged: &n_enc";
run;
filename cl clear;

/* SECTION 5 row-count assertion */
proc sql noprint; select count(*) into :n_prep trimmed from g.prep_mdN; quit;
%macro assert_row_count(actual=, expected=, src=);
  %if &actual ne &expected %then %do; %put ERROR: PREP row count mismatch for &src -- expected &expected got &actual; %abort cancel; %end;
  %else %put NOTE: OK -- &src row count &actual matches expected &expected;
%mend assert_row_count;
%assert_row_count(actual=&n_prep, expected=&expected_nobs, src=mdN);
%put NOTE: ==== Phase 3 prep mdN complete ====;
```

Base_Procedure_1 exists in every source; if PROC CONTENTS (qc/03_charvars_all.txt) shows a

Widths come from qc/03_charvars_all.txt — do NOT guess. Conversion logs go to logs/, exception
reports go to qc/ (different directories, PREP-02 vs PREP-06). LENGTH before SET, always.
