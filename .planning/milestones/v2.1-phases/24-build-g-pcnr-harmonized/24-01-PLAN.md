    SECTION 1 — build `work._recode_rules` (one row per variable, raw_hex, action=MISSING).
    Use PROC SQL, building two tables and stacking them (this supersedes the RESEARCH.md
    lines 462-482 example, which joined wildcards on the blank raw_hex and let a wildcard
    duplicate or override a per-variable row):
    ```sas
    proc sql;
      create table work._rules_pv as                       /* per-variable MISSING, any var_type */
      select d.variable, d.raw_hex, c.raw_value, c.var_type, d.action,
             'PER_VAR' as rule_source length=8, c.n_rows as n_expected
      from work._decisions_raw d
           inner join work._candidates_raw c on d.variable = c.variable and d.raw_hex = c.raw_hex
           inner join work._name_map m       on m.source_name = c.variable and m.role in ('KEEP','KEY')
      where d.variable ne '*' and d.action = 'MISSING';

      create table work._rules_wc as                       /* wildcard expansion */
      select c.variable, c.raw_hex, c.raw_value, c.var_type, d.action,
             'WILDCARD' as rule_source length=8, c.n_rows as n_expected
      from work._decisions_raw d
           inner join work._candidates_raw c on upcase(d.normalized_value) = upcase(c.normalized_value)
           inner join work._name_map m       on m.source_name = c.variable and m.role in ('KEEP','KEY')
      where d.variable = '*' and d.action = 'MISSING'
        and c.var_type = 'char' and c.candidate_class ne 'AMBIGUOUS'
        and not exists (select 1 from work._decisions_raw p          /* per-var row wins, */
                        where p.variable = c.variable              /* whatever its action */
                          and p.raw_hex  = c.raw_hex);

      create table work._recode_rules as
      select * from work._rules_pv
      outer union corr
      select * from work._rules_wc
      order by variable, raw_hex;
    quit;
    ```
