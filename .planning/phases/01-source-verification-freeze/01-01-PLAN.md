    ```sas
    %macro get_sha256(filepath=, outvar=);
      filename _hpipe pipe "certutil -hashfile ""&filepath"" SHA256";
      data _null_;
        infile _hpipe truncover;
        input line $200.;
        line = strip(line);
        if length(line) = 64 then call symputx("&outvar", line, 'G');
      run;
      filename _hpipe clear;
    %mend get_sha256;
    ```
