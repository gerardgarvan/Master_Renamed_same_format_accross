SECTION 3: BUILD BEST-LABEL TABLE
Join work.raw_labels to work.prec_dict on varname = sas_name (left join).
Derive working_label: DICTIONARY if prec_dict.description non-missing, then SAS_LABEL if non-missing, else VARNAME.
Derive label_source as 'DICTIONARY', 'SAS_LABEL', or 'VARNAME' accordingly.

```sas
proc sql noprint;
  create table work.best_labels as
  select l.varname,
         case
           when not missing(d.description) then strip(d.description)
           when not missing(l.sas_label)   then strip(l.sas_label)
           else l.varname
         end as working_label  length=256,
         case
           when not missing(d.description) then 'DICTIONARY'
           when not missing(l.sas_label)   then 'SAS_LABEL'
           else 'VARNAME'
         end as label_source   length=10
  from work.raw_labels as l
  left join work.prec_dict as d
    on l.varname = d.sas_name;
  /* Count label-source tiers for QC */
  select sum(label_source='DICTIONARY') into :n_dict_label  trimmed from work.best_labels;
  select sum(label_source='SAS_LABEL')  into :n_sas_label   trimmed from work.best_labels;
  select sum(label_source='VARNAME')    into :n_varname_fallback trimmed from work.best_labels;
quit;
%put NOTE: [14] Label source breakdown:;
%put NOTE: [14]   DICTIONARY: &n_dict_label;
%put NOTE: [14]   SAS_LABEL : &n_sas_label;
%put NOTE: [14]   VARNAME   : &n_varname_fallback;
```

SECTION 5: COMPUTE PAIRWISE COMPGED SIMILARITY
Use PROC SQL self-join (preferred over indexed macro variables -- cleaner, established in pipeline):

```sas
proc sql;
  create table work.all_pairs as
  select a.varname                    as varname_a    length=32,
         a.working_label              as label_a      length=256,
         a.label_source               as source_a     length=10,
         b.varname                    as varname_b    length=32,
         b.working_label              as label_b      length=256,
         b.label_source               as source_b     length=10,
         /* Normalized COMPGED: 1 - edit_distance / (2 * max_string_length)
            Score = 1.0 means identical; 0.0 means maximally different.
            Division-by-zero guard: when both strings are empty, score = 1
            (identical content -- zero-length) set to missing to flag.      */
         case
           when 2*max(length(strip(upcase(a.working_label))),
                      length(strip(upcase(b.working_label)))) = 0 then .
           else 1 - compged(upcase(strip(a.working_label)),
                            upcase(strip(b.working_label))) /
                    (2 * max(length(strip(upcase(a.working_label))),
                             length(strip(upcase(b.working_label)))))
         end as score_edit
  from work.best_labels as a,
       work.best_labels as b
  where a.varname < b.varname
    /* Exclude pairs where BOTH labels fall back to variable name --
       name similarity was handled by Phase 11; no new information here */
    and not (a.label_source = 'VARNAME' and b.label_source = 'VARNAME')
    /* Apply threshold AFTER computing to capture full score for QC */
    and calculated score_edit >= &threshold
  order by score_edit desc;

  select count(*) into :n_above_threshold trimmed from work.all_pairs;
quit;
%put NOTE: [14] Pairs above threshold (&threshold): &n_above_threshold;
```

SECTION 6: EXCLUDE KNOWN CONCEPT PAIRS AND H_ COLUMNS

proc sql;
  create table work.candidates as
  select p.*
  from work.all_pairs as p
  where
    /* Exclude pairs already handled in 10_concept_profile.sas */
    not exists (
      select 1 from work.known_pairs as k
      where (k.varname_a = p.varname_a and k.varname_b = p.varname_b)
         or (k.varname_a = p.varname_b and k.varname_b = p.varname_a)
    )

    and p.varname_a not in (
      'H_DEATH_FLAG','H_SLEEP_APNEA','H_DIABETES','H_HYPERLIPIDEMIA',

      'H_FRAILTY_SCORE','H_FRAILTY_CATEGORY','H_ORAL_MORPHINE_EQUIV'

    and p.varname_b not in (
      'H_DEATH_FLAG','H_SLEEP_APNEA','H_DIABETES','H_HYPERLIPIDEMIA',
      'H_HYPERTENSION','H_MOVEMENT_DISORDER','H_COGNITIVE_DISORDER',
