proc sql noprint;
  create table work.col_meta as
  select upcase(name) as pcnr_name_u length=32,
         name as pcnr_name length=32,
         type,
         length
  from dictionary.columns

Step 6f -- Compute n_missing_after per column (one PROC SQL loop):
```sas
data work.missing_after;
  length pcnr_name_u $32 n_missing 8;
  stop;
run;

%macro compute_missing_after;
  %local i colname;
  %do i = 1 %to &n_col_meta;
    proc sql noprint;
      select pcnr_name into :colname trimmed
      from work.col_meta (firstobs=&i obs=&i);
    quit;
    proc sql noprint;
      select count(*) - count(&colname) into :n_miss trimmed
      from g.pcnr_analytic_cohort;
    quit;
    proc sql noprint;
      insert into work.missing_after values(upcase("&colname"), &n_miss);
    quit;
  %end;
%mend compute_missing_after;
%compute_missing_after;
```

Step 6g -- Join into work.vars_sheet (one row per column: pcnr_name, source_name, role, label, type, length, n_recoded, n_missing_after):
```sas
proc sql noprint;
  create table work.vars_sheet as
  select c.pcnr_name,
         coalesce(n.source_name, c.pcnr_name) as source_name length=64,
         coalesce(n.role, 'KEEP')              as role        length=8,
         coalesce(n.label, c.pcnr_name)        as label       length=256,
         c.type,
         c.length,
         coalesce(r.n_recoded_total, 0)        as n_recoded,
         coalesce(m.n_missing, 0)              as n_missing_after
  from work.col_meta as c
  left join work.name_map as n
    on upcase(c.pcnr_name) = upcase(n.pcnr_name)
  left join work.recode_totals as r
    on upcase(c.pcnr_name) = upcase(r.varname)
  left join work.missing_after as m
    on upcase(c.pcnr_name) = m.pcnr_name_u
  order by c.pcnr_name;
quit;
```
