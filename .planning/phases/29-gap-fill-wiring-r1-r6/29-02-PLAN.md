Removed-row count assertion (approved count from PCM-D-29 -- prevents silent source file changes):
```sas
%macro r1_removed_gate;
  %let _r1_removed_n = 0;
  proc sql noprint;
    select count(*) into :_r1_removed_n trimmed from work._r1_dup_removed;
  quit;
  %if &_r1_removed_n ne 3 %then %do;
    %fail_out(msg=r1 de-dup removed &_r1_removed_n rows - expected 3 per PCM-D-29 -- source file may have changed);
  %end;
%mend r1_removed_gate;
%r1_removed_gate;
```

Post-dedup duplicate check (second sort -- the deduped dataset must itself be duplicate-free):
```sas
proc sort data=work.r1_normed nodupkey dupout=work._r1_post_check;
  by PRECEDE_STUDY_ID;
run;
%macro r1_post_gate;
  %let _r1_post_n = 0;
  proc sql noprint;
    select count(*) into :_r1_post_n trimmed from work._r1_post_check;
  quit;
  %if &_r1_post_n > 0 %then %do;
    %fail_out(msg=r1 still has &_r1_post_n duplicate PRECEDE_STUDY_ID rows after de-dup -- resolve PCM-D-29 before merge);
  %end;
%mend r1_post_gate;
%r1_post_gate;
```
