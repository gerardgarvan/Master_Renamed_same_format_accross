Pre-merge collision check pattern:

IMPORTANT: Do NOT check g.gapfill_rN columns against g.master_data_merged output from a previous run -- on the second run, previous output already contains the gap-fill columns and the check aborts incorrectly. Instead, check donors against the SNAP baseline (columns that existed before wiring) or cross-check donors against each other.

```sas
%macro check_column_collisions;
  %let _collision_n = 0;
  proc sql noprint;
    /* Check cross-donor collisions: any column name appearing in 2+ gapfill_rN donors? */
    select count(*) into :_collision_n trimmed
    from (
      select upcase(name) as uname, count(distinct memname) as n_donors
      from dictionary.columns
      where libname = 'G'
        and memname in ('GAPFILL_R1','GAPFILL_R2','GAPFILL_R3','GAPFILL_R4','GAPFILL_R5','GAPFILL_R6')
        and upcase(name) ne 'PRECEDE_STUDY_ID'
      group by uname
      having n_donors > 1
    );
  quit;
  %if &_collision_n > 0 %then %do;
    %fail_out(msg=Pre-merge collision check failed -- &_collision_n column name(s) appear in multiple g.gapfill_rN donors -- review gapfill_allowlist.csv);
  %end;

  /* Also check against snap baseline columns (pre-wiring column list -- idempotent across runs) */
  %let _snap_collision_n = 0;
  proc sql noprint;
    select count(*) into :_snap_collision_n trimmed
    from dictionary.columns as d
    inner join dictionary.columns as s
      on upcase(d.name) = upcase(s.name)
    where d.libname = 'G'
      and d.memname in ('GAPFILL_R1','GAPFILL_R2','GAPFILL_R3','GAPFILL_R4','GAPFILL_R5','GAPFILL_R6')
      and s.libname = 'SNAP' and s.memname = 'MASTER_DATA_MERGED'
      and upcase(d.name) ne 'PRECEDE_STUDY_ID';
  quit;
  %if &_snap_collision_n > 0 %then %do;
    %fail_out(msg=Pre-merge collision check failed -- &_snap_collision_n column(s) in g.gapfill_rN already existed in snap baseline -- remove from gapfill_allowlist.csv);
  %end;
  %else %do;
    %put NOTE: [04_merge GAP-03] Pre-merge collision check passed -- no donor columns duplicate baseline or each other;
  %end;
%mend check_column_collisions;
%check_column_collisions;
```

/* -----------------------------------------------------------------------

      select master_data_merged master_data_harmonized;

  %end;

  %else %do;
    /* ---- PASS 2: Snapshot exists -- run PROC COMPARE ---- */
    %put NOTE: [29_gapfill_compare PASS 2] Baseline snapshot found -- running PROC COMPARE;

    /* Derive original column list from snap (excludes new extension columns) */
    proc sql noprint;
      select name into :_orig_merged_cols separated by ' '

      /* sysinfo bit meanings:
           64    = base has ID compare dataset lacks
           128   = compare has ID base lacks
           2048  = compare has variables not in base (EXPECTED after wiring -- new columns)
           4096  = value differences in matched columns
           8192  = variable type changed
