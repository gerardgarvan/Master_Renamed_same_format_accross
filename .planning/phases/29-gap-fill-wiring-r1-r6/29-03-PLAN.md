```sas
%macro ensure_snap_baseline;

  proc sql noprint;
    select count(*) into :_snap_exists trimmed
    from dictionary.members
    where libname='SNAP' and memname='MASTER_DATA_MERGED';

  %if &_snap_exists = 0 %then %do;

    %put NOTE: [04_merge / GAP-03] Existing baseline snapshot found -- not overwritten. Using existing snap.;
  %end;
%mend ensure_snap_baseline;
```

Add the PROC COMPARE block to `sas/04_merge.sas` after the row-count assertion. This block derives the original column list from `snap.master_data_merged` at runtime (not a hardcoded list), runs PROC COMPARE restricted to those columns, and asserts `sysinfo & 4096 = 0`.

```sas
/* ====================================================================
   GAP-03: PROC COMPARE -- verify no value changes in original columns
   base  = snap.master_data_merged (pre-wiring baseline)
   compare = g.master_data_merged  (post-wiring)
   VAR list restricted to original columns (from snap dictionary) to
   prevent sysinfo bit 8 from firing on new extension columns.
   ==================================================================== */

/* Derive original column list from snap (excludes new extension columns) */
proc sql noprint;
  select name into :_orig_merged_cols separated by ' '
  from dictionary.columns
  where libname='SNAP' and memname='MASTER_DATA_MERGED'
    and upcase(name) ne 'PRECEDE_STUDY_ID';  /* ID handled by ID statement */
quit;

proc compare base=snap.master_data_merged
             compare=g.master_data_merged
             noprint;
  id PRECEDE_STUDY_ID;
  var &_orig_merged_cols;
run;

%macro assert_no_value_diffs_merged;
  %if %eval(&sysinfo & 4096) ne 0 %then %do;
    %fail_out(msg=GAP-03 FAILED -- PROC COMPARE found value differences in original columns of g.master_data_merged -- existing rows were altered);
  %end;
  %else %do;
    %put NOTE: [GAP-03] PROC COMPARE PASSED -- g.master_data_merged original columns are byte-identical to baseline snapshot.;
  %end;
%mend assert_no_value_diffs_merged;
%assert_no_value_diffs_merged;

/* Repeat for master_data_harmonized */
proc sql noprint;
  select name into :_orig_harmon_cols separated by ' '
  from dictionary.columns
  where libname='SNAP' and memname='MASTER_DATA_HARMONIZED'
    and upcase(name) ne 'PRECEDE_STUDY_ID';
quit;

proc compare base=snap.master_data_harmonized
             compare=g.master_data_harmonized
             noprint;
  id PRECEDE_STUDY_ID;
  var &_orig_harmon_cols;
run;

%macro assert_no_value_diffs_harmon;
  %if %eval(&sysinfo & 4096) ne 0 %then %do;

  %end;
%mend assert_no_value_diffs_harmon;
%assert_no_value_diffs_harmon;
```

- g.master_data_harmonized: if the harmonized dataset is also extended with new columns in Plan 3, the same PROC COMPARE logic applies. If harmonized is NOT extended in this plan, the PROC COMPARE will confirm it is unchanged. Either outcome is valid.
- `%fail_out` is already defined in the program from earlier sections — do NOT redefine it.
