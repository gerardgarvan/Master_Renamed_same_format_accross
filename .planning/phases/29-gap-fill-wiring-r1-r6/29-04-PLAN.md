```sas
/* Gate: abort if any wired column in g.master_data_harmonized lacks PCNR_APPROVED=1 */
%macro pcnr_gate_new_cols;
  %let _n_unapproved = 0;
  proc sql noprint;
    select count(*) into :_n_unapproved trimmed
    from dictionary.columns as c
    left join work._pcnr_map as m
      on upcase(c.name) = upcase(m.source_name)
    where c.libname = 'G' and c.memname = 'MASTER_DATA_HARMONIZED'
      and m.source_name is null;  /* column in dataset but not in map */
  quit;
  %if &_n_unapproved > 0 %then %do;
    %fail_out(msg=PCNR gate FAILED -- &_n_unapproved column(s) in g.master_data_harmonized have no pcnr_name_map.csv entry -- review sentinel_decisions.csv and set PCNR_APPROVED=1);
  %end;
%mend pcnr_gate_new_cols;
%pcnr_gate_new_cols;
```
